<script>
  const { useState } = React;

  // Image Gallery Component
  function Gallery() {
    const images = [
      { src: 'https://via.placeholder.com/200?text=Nature1', category: 'nature' },
      { src: 'https://via.placeholder.com/200?text=Nature2', category: 'nature' },
      { src: 'https://via.placeholder.com/200?text=City1', category: 'city' },
      { src: 'https://via.placeholder.com/200?text=City2', category: 'city' },
      { src: 'https://via.placeholder.com/200?text=Animal1', category: 'animal' },
      { src: 'https://via.placeholder.com/200?text=Animal2', category: 'animal' }
    ];

    const [selectedCategory, setSelectedCategory] = useState('all');

    // Filter images by category
    const filteredImages = selectedCategory === 'all' ? images : images.filter(image => image.category === selectedCategory);

    return (
      React.createElement('div', { className: 'gallery-container' },
        React.createElement('div', { className: 'category-buttons' },
          React.createElement('button', { onClick: () => setSelectedCategory('all') }, 'All'),
          React.createElement('button', { onClick: () => setSelectedCategory('nature') }, 'Nature'),
          React.createElement('button', { onClick: () => setSelectedCategory('city') }, 'City'),
          React.createElement('button', { onClick: () => setSelectedCategory('animal') }, 'Animals')
        ),
        React.createElement('div', { className: 'grid' },
          filteredImages.map((image, index) =>
            React.createElement('img', { key: index, src: image.src, alt: image.category })
          )
        )
      )
    );
  }

  const root = ReactDOM.createRoot(document.getElementById('root'));
  root.render(React.createElement(Gallery));
</script>
