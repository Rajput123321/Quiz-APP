<script>
  const questions = document.querySelectorAll('.question');
  const nextBtn = document.getElementById('nextBtn');
  const prevBtn = document.getElementById('prevBtn');
  const resultDiv = document.getElementById('result');
  const correctAnswers = [1, 1, 1, 1, 1]; // correct option index for each question
  let current = 0;

  function showQuestion(index) {
    questions.forEach((q, i) => {
      q.classList.toggle('active', i === index);
    });
    prevBtn.disabled = index === 0;
    nextBtn.textContent = index === questions.length - 1 ? 'Submit' : 'Next';
  }

  function getScore() {
    let score = 0;
    correctAnswers.forEach((correct, i) => {
      const answer = document.querySelector(`input[name="q${i}"]:checked`);
      if (answer && parseInt(answer.value) === correct) {
        score++;
      }
    });
    return score;
  }

  nextBtn.onclick = () => {
    if (current < questions.length - 1) {
      current++;
      showQuestion(current);
    } else {
      const score = getScore();
      resultDiv.textContent = `You scored ${score} out of ${questions.length}`;
      resultDiv.style.display = 'block';
      nextBtn.style.display = 'none';
      prevBtn.style.display = 'none';
      questions.forEach(q => q.style.display = 'none');
    }
  };

  prevBtn.onclick = () => {
    if (current > 0) {
      current--;
      showQuestion(current);
    }
  };

  showQuestion(current);
</script>
