<script>
  const { useState } = React;

  // CityWeather Component
  function CityWeather(props) {
    const { city, temperature, description, humidity } = props;
    return (
      React.createElement('div', { className: 'weather-container' },
        React.createElement('h1', null, `Weather in ${city}`),
        React.createElement('div', { className: 'temperature' }, `${temperature}°C`),
        React.createElement('div', { className: 'details' }, `Description: ${description}`),
        React.createElement('div', { className: 'details' }, `Humidity: ${humidity}%`)
      )
    );
  }

  // WeatherApp Component
  function WeatherApp() {
    const weatherData = {
      city: 'New York',
      temperature: 25,  // Hardcoded temperature in °C
      description: 'Sunny',
      humidity: 60  // Hardcoded humidity percentage
    };

    return (
      React.createElement(CityWeather, weatherData)
    );
  }

  const root = ReactDOM.createRoot(document.getElementById('root'));
  root.render(React.createElement(WeatherApp));
</script>
