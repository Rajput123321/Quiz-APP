<script>
  const studentForm = document.getElementById('studentForm');
  const nameInput = document.getElementById('name');
  const rollNoInput = document.getElementById('rollNo');
  const gradeSelect = document.getElementById('grade');
  const studentsTable = document.getElementById('studentsTable').getElementsByTagName('tbody')[0];
  const editIndexInput = document.getElementById('editIndex');

  let students = [];

  // Function to render the student table
  function renderTable() {
    studentsTable.innerHTML = '';
    students.forEach((student, index) => {
      const row = studentsTable.insertRow();
      row.innerHTML = `
        <td>${student.name}</td>
        <td>${student.rollNo}</td>
        <td>${student.grade}</td>
        <td>
          <button class="edit-btn" onclick="editStudent(${index})">Edit</button>
          <button class="delete-btn" onclick="deleteStudent(${index})">Delete</button>
        </td>
      `;
    });
  }

  // Add or Edit student record
  studentForm.addEventListener('submit', function (e) {
    e.preventDefault();

    const name = nameInput.value.trim();
    const rollNo = rollNoInput.value.trim();
    const grade = gradeSelect.value;

    if (editIndexInput.value !== '') {
      // Edit existing student
      const index = editIndexInput.value;
      students[index] = { name, rollNo, grade };
      editIndexInput.value = '';  // Reset the form for new entries
    } else {
      // Add new student
      students.push({ name, rollNo, grade });
    }

    // Reset form
    nameInput.value = '';
    rollNoInput.value = '';
    gradeSelect.value = 'A';

    renderTable();
  });

  // Edit student record
  function editStudent(index) {
    const student = students[index];
    nameInput.value = student.name;
    rollNoInput.value = student.rollNo;
    gradeSelect.value = student.grade;
    editIndexInput.value = index;
  }

  // Delete student record
  function deleteStudent(index) {
    students.splice(index, 1);
    renderTable();
  }

</script>
