  <h2>Shopping Cart</h2>

  <!-- Cart Items List -->
  <div id="cartItems"></div>

  <!-- Total Price Display -->
  <div class="total" id="totalPrice">Total: $0.00</div>
</div>

<script>
  // Sample items available for purchase
  const items = [
    { id: 1, name: "Laptop", price: 799.99 },
    { id: 2, name: "Phone", price: 599.99 },
    { id: 3, name: "Headphones", price: 99.99 },
    { id: 4, name: "Keyboard", price: 49.99 }
  ];

  // Initialize empty cart
  let cart = [];

  // Function to render cart items
  function renderCart() {
    const cartItemsContainer = document.getElementById('cartItems');
    cartItemsContainer.innerHTML = '';  // Clear previous items

    cart.forEach(item => {
      const itemElement = document.createElement('div');
      itemElement.className = 'item';
      itemElement.innerHTML = `
        <div class="item-name">${item.name}</div>
        <div class="item-price">$${item.price.toFixed(2)}</div>
        <div class="item-actions">
          <button class="remove-btn" onclick="removeItem(${item.id})">Remove</button>
        </div>
      `;
      cartItemsContainer.appendChild(itemElement);
    });

    updateTotal();
  }

  // Function to update total price
  function updateTotal() {
    const totalPrice = cart.reduce((sum, item) => sum + item.price, 0);
    document.getElementById('totalPrice').textContent = `Total: $${totalPrice.toFixed(2)}`;
  }

  // Function to add item to cart
  function addItemToCart(itemId) {
    const item = items.find(i => i.id === itemId);
    if (item) {
      cart.push(item);
      renderCart();
    }
  }

  // Function to remove item from cart
  function removeItem(itemId) {
    cart = cart.filter(item => item.id !== itemId);
    renderCart();
  }

  // Render sample items and add to cart buttons
  items.forEach(item => {
    const itemElement = document.createElement('div');
    itemElement.className = 'item';
    itemElement.innerHTML = `
      <div class="item-name">${item.name}</div>
      <div class="item-price">$${item.price.toFixed(2)}</div>
      <div class="item-actions">
        <button class="add-btn" onclick="addItemToCart(${item.id})">Add to Cart</button>
      </div>
    `;
    document.querySelector('.container').insertBefore(itemElement, document.getElementById('cartItems'));
  });
</script>
